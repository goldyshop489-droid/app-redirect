<!DOCTYPE html>

<html>

<head>

  <title>App Redirect</title>


  <script>

    function redirectUser() {

      var ua = navigator.userAgent || navigator.vendor || window.opera;


      // Android

      if (/android/i.test(ua)) {

        window.location.href =

          "https://play.google.com/store/apps/details?id=com.easybrain.jigsaw.puzzles&hl=en_IN";

      }


      // iPhone / iPad

      else if (/iPad|iPhone|iPod/.test(ua) && !window.MSStream) {

        window.location.href =

          "https://apps.apple.com/us/app/jigsaw-puzzles-puzzle-games/id1324604053";

      }


      // Desktop fallback

      else {

        window.location.href =

          "https://easybrain.com";

      }

    }

  </script>

</head>


<body onload="redirectUser()">

  <h2>Redirecting...</h2>

</body>

</html>
